        // Render Home Data
        function renderHome() {
            const currentYear = new Date().getFullYear();
            let totalInc = 0, totalExp = 0;

            appData.transactions.forEach(tx => {
                const txYear = new Date(tx.date).getFullYear();
                if (txYear === currentYear) {
                    if (tx.type === 'income') totalInc += tx.amount;
                    if (tx.type === 'expense') totalExp += tx.amount;
                }
            });

            const totalSavings = totalInc - totalExp;
            const savingsRate = totalInc > 0 ? Math.max(0, Math.round((totalSavings / totalInc) * 100)) : 0;

            document.getElementById('totalYearIncome').innerText = `฿${totalInc.toLocaleString()}`;
            document.getElementById('totalYearExpense').innerText = `฿${totalExp.toLocaleString()}`;
            document.getElementById('totalYearSavings').innerText = `฿${totalSavings.toLocaleString()}`;

            // Render Monthly Budget Progress Bar
            const now = new Date();
            const currentMonthExp = appData.transactions.filter(tx => {
                const d = new Date(tx.date);
                return d.getMonth() === now.getMonth() && d.getFullYear() === now.getFullYear() && tx.type === 'expense';

        function renderSummary() {

        function renderCalendarGrid(m, y) {

            const daysInMonth = new Date(y, m + 1, 0).getDate();

            for (let i = 0; i < firstDay; i++) {
                grid.appendChild(document.createElement('div'));
            }

            for (let day = 1; day <= daysInMonth; day++) {
                const dayExp = appData.transactions.filter(tx => {
                    const d = new Date(tx.date);
                    return d.getDate() === day && d.getMonth() === m && d.getFullYear() === y && tx.type === 'expense';
                }).reduce((sum, tx) => sum + tx.amount, 0);

                const cell = document.createElement('div');
                cell.className = 'p-1 rounded-xl bg-darkBg border border-slate-800 flex flex-col items-center justify-between min-h-[42px]';
                cell.innerHTML = `
                    <span class="text-[9px] text-slate-400 font-bold">${day}</span>
                    ${dayExp > 0 ? `<span class="text-[8px] text-neonPink font-bold truncate w-full privacy-target">฿${dayExp}</span>` : ''}
                `;
                grid.appendChild(cell);
            }
        }

        function renderYearlyInsights() {
            const currentYear = new Date().getFullYear();
            const monthlySums = new Array(12).fill(0);
            let totalCount = 0;
            let totalExpYear = 0;
            const catMap = {};
            const dayOfWeekMap = new Array(7).fill(0);

            appData.transactions.forEach(tx => {
                const d = new Date(tx.date);
                if (d.getFullYear() === currentYear && tx.type === 'expense') {
                    monthlySums[d.getMonth()] += tx.amount;
                    totalCount++;
                    totalExpYear += tx.amount;
                    catMap[tx.categoryName] = (catMap[tx.categoryName] || 0) + tx.amount;
                    dayOfWeekMap[d.getDay()] += tx.amount;
                }
            });

            if (barChartInstance) barChartInstance.destroy();

            const ctx = document.getElementById('yearlyBarChart').getContext('2d');
            barChartInstance = new Chart(ctx, {
                type: 'bar',
                data: {
                    labels: ['ม.ค.', 'ก.พ.', 'มี.ค.', 'เม.ย.', 'พ.ค.', 'มิ.ย.', 'ก.ค.', 'ส.ค.', 'ก.ย.', 'ต.ค.', 'พ.ย.', 'ธ.ค.'],
                    datasets: [{
                        data: monthlySums,
                        backgroundColor: '#A855F7',
                        borderRadius: 6
                    }]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    plugins: { legend: { display: false } },
                    scales: {
                        x: { ticks: { color: '#94A3B8', font: { size: 9 } } },
                        y: { ticks: { color: '#94A3B8', font: { size: 9 } } }
                    }
                }
            });

            document.getElementById('statTotalCount').innerText = `${totalCount} รายการ`;
            document.getElementById('statAvgPerDay').innerText = `฿${Math.round(totalExpYear / 365).toLocaleString()}`;

            const days = ['อาทิตย์', 'จันทร์', 'อังคาร', 'พุธ', 'พฤหัสบดี', 'ศุกร์', 'เสาร์'];
            const maxDayIdx = dayOfWeekMap.indexOf(Math.max(...dayOfWeekMap));
            document.getElementById('statMaxDayOfWeek').innerText = totalExpYear > 0 ? days[maxDayIdx] : '-';

            const months = ['มกราคม', 'กุมภาพันธ์', 'มีนาคม', 'เมษายน', 'พฤษภาคม', 'มิถุนายน', 'กรกฎาคม', 'สิงหาคม', 'กันยายน', 'ตุลาคม', 'พฤศจิกายน', 'ธันวาคม'];
            const maxMonthIdx = monthlySums.indexOf(Math.max(...monthlySums));
            document.getElementById('statMaxMonth').innerText = totalExpYear > 0 ? months[maxMonthIdx] : '-';

            let topCat = '-', topCatAmt = 0;
            Object.keys(catMap).forEach(cat => {
                if (catMap[cat] > topCatAmt) {
                    topCatAmt = catMap[cat];
                    topCat = cat;
                }
            });
            document.getElementById('statTopCatName').innerText = topCat;
            document.getElementById('statTopCatAmount').innerText = `฿${topCatAmt.toLocaleString()}`;
        }

        // OCR Slip Scanner
        function processSlipOCR(event) {
            const file = event.target.files[0];
            if (!file) return;

            Tesseract.recognize(file, 'tha+eng', { logger: m => console.log(m) })
                .then(({ data: { text } }) => {
                    const amountMatch = text.match(/(\d{1,3}(,\d{3})*(\.\d{2})?)\s*(บาท|THB)/i) || text.match(/จำนวนเงิน\s*(\d+(\.\d{2})?)/);
                    if (amountMatch) {
                        const cleanAmount = amountMatch[1].replace(',', '');
                        document.getElementById('txAmount').value = cleanAmount;
                    }
                    document.getElementById('txTitle').value = 'สแกนสลิปโอนเงิน';
                    alert('สแกนอ่านข้อมูลสลิปเรียบร้อยแล้ว!');
                })
                .catch(err => {
                    alert('ไม่สามารถอ่านข้อมูลจากสลิปได้ กรุณากรอกด้วยตนเอง');
                });
        }

        function saveLoan() {
            const person = document.getElementById('loanPerson').value.trim();
            const amount = parseFloat(document.getElementById('loanAmount').value);

            if (!person || !amount) {
                alert('กรุณากรอกข้อมูลให้ครบถ้วน');
                return;
            }

            appData.loans.push({
                id: 'loan_' + Date.now(),
                type: currentLoanType,
                person: person,
                amount: amount,
                isSettled: false,
                date: new Date().toISOString().split('T')[0]
            });

            saveData();
            closeModal('addLoanModal');
            document.getElementById('loanPerson').value = '';
            document.getElementById('loanAmount').value = '';
            renderLoans();
        }

        function renderLoans() {
            const list = document.getElementById('loanList');
            list.innerHTML = '';

            let totalLent = 0, totalBorrowed = 0;

            appData.loans.forEach(loan => {
                if (!loan.isSettled) {
                    if (loan.type === 'lent') totalLent += loan.amount;
                    if (loan.type === 'borrowed') totalBorrowed += loan.amount;
                }

                const card = document.createElement('div');
                card.className = `glass-card p-3 rounded-2xl flex items-center justify-between border ${loan.isSettled ? 'border-slate-800 opacity-60' : 'border-slate-700'}`;
                card.innerHTML = `
                    <div class="flex items-center gap-3">
                        <div class="w-8 h-8 rounded-xl flex items-center justify-center text-xs ${loan.type === 'lent' ? 'bg-neonCyan/20 text-neonCyan' : 'bg-neonYellow/20 text-neonYellow'}">
                            <i class="fa-solid ${loan.type === 'lent' ? 'fa-hand-holding-hand' : 'fa-hand-holding-dollar'}"></i>
                        </div>
                        <div>
                            <p class="text-xs font-bold text-slate-200">${loan.person} <span class="text-[9px] text-slate-400">(${loan.type === 'lent' ? 'ให้ยืม' : 'ยืมเขา'})</span></p>
                            <p class="text-[10px] text-slate-400">${loan.date}</p>
                        </div>
                    </div>
                    <div class="text-right flex items-center gap-2">
                        <div>
                            <p class="text-xs font-bold ${loan.type === 'lent' ? 'text-neonCyan' : 'text-neonYellow'} privacy-target">฿${loan.amount.toLocaleString()}</p>
                            <p class="text-[9px] ${loan.isSettled ? 'text-neonGreen' : 'text-slate-500'}">${loan.isSettled ? 'คืนแล้ว' : 'ค้างชำระ'}</p>
                        </div>
                        ${!loan.isSettled ? `<button onclick="settleLoan('${loan.id}')" class="px-2 py-1 bg-neonGreen/20 text-neonGreen border border-neonGreen/40 rounded-lg text-[10px] font-bold">ยืนยันคืน</button>` : ''}
                    </div>
                `;
                list.appendChild(card);
            });

            document.getElementById('totalLent').innerText = `฿${totalLent.toLocaleString()}`;
            document.getElementById('totalBorrowed').innerText = `฿${totalBorrowed.toLocaleString()}`;
        }

        function settleLoan(id) {
            const loan = appData.loans.find(l => l.id === id);
            if (loan) {
                loan.isSettled = true;
                saveData();
                renderLoans();
            }
        }

            const bill = appData.bills.find(b => b.id === id);

                appData.transactions.push({
                    id: 'tx_' + Date.now(),
                    type: 'expense',
                    amount: bill.amount,
                    title: `ชำระบิล: ${bill.title}`,
                    date: new Date().toISOString().split('T')[0],
                    categoryId: 'exp5',
                    categoryName: 'ค่าสาธารณูปโภค',
                    categoryIcon: 'fa-bolt',
                    categoryColor: '#F59E0B'
                });
                saveData();
                renderHome();
                alert('ลงบันทึกการชำระบิลเรียบร้อย!');
            }
        }

        function exportCSV() {
            let csv = '\uFEFFDate,Type,Title,Category,Amount\n';
            appData.transactions.forEach(tx => {
                csv += `"${tx.date}","${tx.type}","${tx.title}","${tx.categoryName}",${tx.amount}\n`;
            });

            const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' });
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url;
            a.download = `expense_backup_${new Date().toISOString().split('T')[0]}.csv`;
            a.click();
        }

        function exportJSON() {
            const blob = new Blob([JSON.stringify(appData, null, 2)], { type: 'application/json' });
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url;
            a.download = `expense_data_${new Date().toISOString().split('T')[0]}.json`;
            a.click();
        }

        function importJSON(event) {
            const file = event.target.files[0];
            if (!file) return;

            const reader = new FileReader();
            reader.onload = (e) => {
                try {
                    const imported = JSON.parse(e.target.result);
                    appData = { ...appData, ...imported };
                    saveData();
                    renderHome();
                    closeModal('settingsModal');
                    alert('นำเข้าข้อมูลเรียบร้อยแล้ว!');
                } catch(err) { alert('ไฟล์ JSON ไม่ถูกต้อง'); }
            };
            reader.readAsText(file);
        }
